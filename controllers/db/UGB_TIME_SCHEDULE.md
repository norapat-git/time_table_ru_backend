DROP TABLE NORAPAT.UGB_TIME_SCHEDULE CASCADE CONSTRAINTS;

CREATE TABLE NORAPAT.UGB_TIME_SCHEDULE
(
  TIME_CODE     NUMBER(2),
  TIME_START    CHAR(4 BYTE),
  TIME_END      CHAR(4 BYTE),
  TIME_RU30     VARCHAR2(50 BYTE),
  FLAG_DISPLAY  NUMBER(1)
)
TABLESPACE SYSTEM
PCTUSED    40
PCTFREE    10
INITRANS   1
MAXTRANS   255
STORAGE    (
            INITIAL          64K
            NEXT             1M
            MINEXTENTS       1
            MAXEXTENTS       UNLIMITED
            PCTINCREASE      0
            FREELISTS        1
            FREELIST GROUPS  1
            BUFFER_POOL      DEFAULT
           )
LOGGING 
NOCOMPRESS 
NOCACHE;

COMMENT ON COLUMN NORAPAT.UGB_TIME_SCHEDULE.TIME_CODE IS 'รหัสเวลา';

COMMENT ON COLUMN NORAPAT.UGB_TIME_SCHEDULE.TIME_START IS 'เวลาเริ่มต้น';

COMMENT ON COLUMN NORAPAT.UGB_TIME_SCHEDULE.TIME_RU30 IS 'เวลาใน มร.30';

COMMENT ON COLUMN NORAPAT.UGB_TIME_SCHEDULE.FLAG_DISPLAY IS 'หมายเหตุ';
