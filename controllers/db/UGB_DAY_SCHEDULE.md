DROP TABLE NORAPAT.UGB_DAY_SCHEDULE CASCADE CONSTRAINTS;

CREATE TABLE NORAPAT.UGB_DAY_SCHEDULE
(
  DAY_CODE           NUMBER(2),
  DAY_NAME_S         VARCHAR2(10 BYTE),
  DAY_NAME_L         VARCHAR2(30 BYTE),
  DAY_NAME_STANDARD  CHAR(3 BYTE)
)
TABLESPACE SYSTEM
PCTUSED    40
PCTFREE    10
INITRANS   1
MAXTRANS   255
STORAGE    (
            INITIAL          64K
            NEXT             1M
            MINEXTENTS       1
            MAXEXTENTS       UNLIMITED
            PCTINCREASE      0
            FREELISTS        1
            FREELIST GROUPS  1
            BUFFER_POOL      DEFAULT
           )
LOGGING 
NOCOMPRESS 
NOCACHE;

COMMENT ON COLUMN NORAPAT.UGB_DAY_SCHEDULE.DAY_CODE IS 'รหัสวันที่';

COMMENT ON COLUMN NORAPAT.UGB_DAY_SCHEDULE.DAY_NAME_S IS 'วันที่แบบย่อ';

COMMENT ON COLUMN NORAPAT.UGB_DAY_SCHEDULE.DAY_NAME_L IS 'วันที่แบบเต็ม';

COMMENT ON COLUMN NORAPAT.UGB_DAY_SCHEDULE.DAY_NAME_STANDARD IS 'วันที่แบบย่อมาตรฐาน';
